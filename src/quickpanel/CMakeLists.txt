cmake_minimum_required(VERSION 3.20)
project(quickpanel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network DBus)
find_package(KF6NetworkManagerQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse libpulse-mainloop-glib)
add_executable(quickpanel main.cpp speedtest.cpp battery.cpp bluetooth.cpp volume.cpp sni.cpp)
target_compile_options(quickpanel PRIVATE -O2 -march=native -Wall -Wextra)
target_compile_definitions(quickpanel PRIVATE QT_USE_QSTRINGBUILDER)
target_link_libraries(quickpanel PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus KF6::NetworkManagerQt PkgConfig::PULSE)
target_link_options(quickpanel PRIVATE -Wl,--as-needed -s)
install(TARGETS quickpanel DESTINATION bin)
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink quickpanel \$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/bin/wifi-panel)")
